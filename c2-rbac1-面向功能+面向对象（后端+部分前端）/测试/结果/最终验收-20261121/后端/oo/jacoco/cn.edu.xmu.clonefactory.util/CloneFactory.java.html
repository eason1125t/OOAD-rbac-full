package cn.edu.xmu.clonefactory.util;

import organization.Organization;
import organization.vo.CatalogVO;
import organization.vo.OrganizationVO;
import rolepermission.CatalogEntry;
import simulation.BusinessRequestService;
import simulation.domain.BusinessRequest;
import simulation.vo.RequestDetailVO;
import simulation.vo.RequestSummaryVO;

public final class CloneFactory {
  /**
   * Copy all fields from source to target
   * generated by @CopyFrom(BusinessRequest) in {RequestSummary}
   * @param target the target object
   * @param source the source object
   * @return the copied target object
   */
  public static BusinessRequestService.RequestSummary copy(
      BusinessRequestService.RequestSummary target, BusinessRequest source) {
    target.setId(source.getId());
    target.setTitle(source.getTitle());
    target.setStatus(source.getStatus());
    return target;
  }

  /**
   * Copy all fields from source to target
   * generated by @CopyFrom(BusinessRequest) in {RequestDetail}
   * @param target the target object
   * @param source the source object
   * @return the copied target object
   */
  public static BusinessRequestService.RequestDetail copy(
      BusinessRequestService.RequestDetail target, BusinessRequest source) {
    target.setStatus(source.getStatus());
    target.setTitle(source.getTitle());
    target.setId(source.getId());
    target.setApplicantId(source.getApplicantId());
    return target;
  }

  /**
   * Copy all fields from source to target
   * generated by @CopyFrom(RequestDetail) in {RequestDetailVO}
   * @param target the target object
   * @param source the source object
   * @return the copied target object
   */
  public static RequestDetailVO copy(RequestDetailVO target,
      BusinessRequestService.RequestDetail source) {
    target.setApplicantId(source.getApplicantId());
    target.setStatus(source.getStatus());
    target.setId(source.getId());
    target.setTitle(source.getTitle());
    return target;
  }

  /**
   * Copy all fields from source to target
   * generated by @CopyFrom(Organization) in {OrganizationVO}
   * @param target the target object
   * @param source the source object
   * @return the copied target object
   */
  public static OrganizationVO copy(OrganizationVO target, Organization source) {
    target.setCode(source.getCode());
    target.setName(source.getName());
    target.setSourceParentName(source.getSourceParentName());
    target.setManagerName(source.getManagerName());
    target.setParentId(source.getParentId());
    target.setId(source.getId());
    return target;
  }

  /**
   * Copy all fields from source to target
   * generated by @CopyFrom(CatalogEntry) in {CatalogVO}
   * @param target the target object
   * @param source the source object
   * @return the copied target object
   */
  public static CatalogVO copy(CatalogVO target, CatalogEntry source) {
    target.setProjectManagerCell(source.getProjectManagerCell());
    target.setModuleName(source.getModuleName());
    target.setExecutiveCell(source.getExecutiveCell());
    target.setAdministratorCell(source.getAdministratorCell());
    target.setFeatureName(source.getFeatureName());
    target.setId(source.getId());
    target.setSystemName(source.getSystemName());
    target.setEmployeeCell(source.getEmployeeCell());
    target.setDescription(source.getDescription());
    target.setDepartmentManagerCell(source.getDepartmentManagerCell());
    return target;
  }

  /**
   * Copy all fields from source to target
   * generated by @CopyFrom(RequestSummary) in {RequestSummaryVO}
   * @param target the target object
   * @param source the source object
   * @return the copied target object
   */
  public static RequestSummaryVO copy(RequestSummaryVO target,
      BusinessRequestService.RequestSummary source) {
    target.setTitle(source.getTitle());
    target.setStatus(source.getStatus());
    target.setId(source.getId());
    return target;
  }
}
